Character Move Set PSEUDOCODE
import pygame
import character stats

define walk:
    if arrow key is pressed, use speed in direction with correct frames
    else: wait

define sprint:
    if not walk: 
        wait
    else if walk:
        if X is pressed, increase speed, show sprint frames. 
        else: wait

define attack_enemy:
    if Z is pressed:
        attack frames and hitbox
        if in enemy hitbox: hit enemy
        else: nothing
        if Z is held: 
            attack 2 frames and hitbox
            if in enemy hitbox: hit enemy
            else: nothing
            if Z is still held: 
                attack 3 frames and hitbox
                if in enemy hitbox: hit enemy
                else: nothing
